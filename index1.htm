<script>

    let currentStudent = localStorage.getItem('currentStudent') || '';

    let currentProblem = null;

    let attempts = 0;

 

    // Tananyag feladat sablonok

    const problemTemplates = [

        {

            type: 'ohm',

            title: 'Ohm-törvénye (Feszültség és Áramerősség)',

            gen: (diff) => {

                let U = Math.floor(Math.random() * 20 + 1) * 10; // 10V - 200V

                let R = Math.floor(Math.random() * 10 + 1) * 5;  // 5 Ohm - 50 Ohm

                let I = U / R;

 

                if (diff === 'easy') {

                    return {

                        text: `Egy elektromos fűtőtest ellenállása R = ${R} Ω. Mekkora áramerősség folyik át rajta, ha U = ${U} V feszültségre csatlakoztatjuk?`,

                        data: { U: U, R: R },

                        inputs: [],

                        target: I,

                        unit: 'A',

                        label: 'Áramerősség (I) amperben:'

                    };

                } else if (diff === 'medium') {

                    return {

                        text: `Egy áramkörben lévő fogyasztó ellenállása ${R} Ω. A mérőműszer szerint ${U} V feszültség kapcsolódik rá. Számítsd ki az áramerősséget!`,

                        data: {},

                        inputs: [

                            { key: 'U', name: 'Feszültség (U)', correct: U, unit: 'V' },

                            { key: 'R', name: 'Ellenállás (R)', correct: R, unit: 'Ω' }

                        ],

                        target: I,

                        unit: 'A',

                        label: 'Kiszámított áramerősség (I) A-ben:'

                    };

                } else { // hard

                    let I_mA = I * 1000;

                    return {

                        text: `Egy elektronikai alkatrészen ${U} V feszültség hatására ${I_mA} mA áramerősség folyik át. Számítsd ki az alkatrész ellenállását! (Figyelj a mértékegység-átváltásra!)`,

                        data: {},

                        inputs: [

                            { key: 'U', name: 'Feszültség (U)', correct: U, unit: 'V' },

                            { key: 'I', name: 'Áramerősség A-ben (I)', correct: I, unit: 'A' }

                        ],

                        target: R,

                        unit: 'Ω',

                        label: 'Kiszámított ellenállás (R) Ω-ban:'

                    };

                }

            }

        },

        {

            type: 'power',

            title: 'Elektromos Teljesítmény',

            gen: (diff) => {

                let U = 230;

                let I = Math.floor(Math.random() * 10 + 1); // 1A - 10A

                let P = U * I;

 

                if (diff === 'easy') {

                    return {

                        text: `Egy háztartási eszköz a 230 V-os hálózatra kapcsolva ${I} A áramot vesz fel. Mekkora az elektromos teljesítménye?`,

                        data: { U: 230, I: I },

                        inputs: [],

                        target: P,

                        unit: 'W',

                        label: 'Teljesítmény (P) Wattban:'

                    };

                } else if (diff === 'medium') {

                    return {

                        text: `A konyhai vízforraló 230 V-os feszültségen működik, és ${I} A áramerősséget vesz fel a hálózatból. Mekkora a készülék teljesítménye?`,

                        data: {},

                        inputs: [

                            { key: 'U', name: 'Feszültség (U)', correct: 230, unit: 'V' },

                } else { // hard

                    return {

                        text: `Egy elektromos sütő teljesítménye ${P_kW} kW. Mekkora áramot vesz fel a hálózatból, ha a hálózati feszültség 230 V?`,

                        data: {},

                        inputs: [

                            { key: 'P', name: 'Teljesítmény W-ban (P)', correct: P, unit: 'W' },

                            { key: 'U', name: 'Feszültség (U)', correct: 230, unit: 'V' }

                        ],

                        target: I,

                        unit: 'A',

                        label: 'Kiszámított áramerősség (I) A-ben:'

                    };

                }

            }

        }

    ];

 

    window.onload = function() {

        if (currentStudent) {

            showPractice();

        }

    };

 

    function registerStudent() {

        const name = document.getElementById('student-name').value.trim();

        if (!name) { alert('Kérlek, add meg a neved!'); return; }

        currentStudent = name;

        localStorage.setItem('currentStudent', currentStudent);

        showPractice();

    }

 

    function showPractice() {

        document.getElementById('user-display').innerText = '👤 Diák: ' + currentStudent;

        document.getElementById('login-section').style.display = 'none';

        document.getElementById('practice-section').style.display = 'block';

        generateProblem();

    }

 

    function switchTab(tabId) {

        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

        document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));

 

        event.target.classList.add('active');

        document.getElementById(tabId).classList.add('active');

    }

 

    function generateProblem() {

        attempts = 0;

        document.getElementById('retry-count').innerText = 'Próbálkozások: 0';

        document.getElementById('feedback-box').className = 'feedback';

        document.getElementById('student-answer').value = '';

 

        const diff = document.getElementById('difficulty-select').value;

        const badge = document.getElementById('problem-badge');

 

        if (diff === 'easy') {

            badge.innerText = '🟢 Könnyű Szint';

            badge.className = 'badge badge-easy';

        } else if (diff === 'medium') {

            badge.innerText = '🟡 Közepes Szint';

            badge.className = 'badge badge-medium';

        } else {

            badge.innerText = '🔴 Nehéz Szint';

            badge.className = 'badge badge-hard';

        }

 

        const template = problemTemplates[Math.floor(Math.random() * problemTemplates.length)];

        currentProblem = template.gen(diff);

 

        document.getElementById('problem-text').innerText = currentProblem.text;

        document.getElementById('answer-label').innerText = currentProblem.label;

 

        // Nézetek frissítése nehézség szerint

        const easyBox = document.getElementById('easy-data-container');

        const medHardBox = document.getElementById('medium-hard-data-container');

 

        if (diff === 'easy') {

            easyBox.style.display = 'block';

            medHardBox.style.display = 'none';

            let html = '';

            for (let k in currentProblem.data) {

                html += `<div>• ${k} = ${currentProblem.data[k]}</div>`;

            }

            document.getElementById('easy-data-list').innerHTML = html;

        } else {

            easyBox.style.display = 'none';

            medHardBox.style.display = 'block';

            let html = '';

            currentProblem.inputs.forEach((inp, idx) => {

                html += `<div class="form-group">

                            <label>${inp.name} (${inp.unit}):</label>

                            <input type="number" id="data-input-${idx}" placeholder="Írd ki a szövegből...">

                         </div>`;

            });

            document.getElementById('data-inputs').innerHTML = html;

        }

    }

 

    function checkAnswer() {

        attempts++;

        document.getElementById('retry-count').innerText = `Próbálkozások: ${attempts}`;

        const diff = document.getElementById('difficulty-select').value;

        const feedback = document.getElementById('feedback-box');

 

        // 1. Adatkiírás ellenőrzése (Közepes és Nehéz szinten)

        if (diff !== 'easy') {

            for (let i = 0; i < currentProblem.inputs.length; i++) {

                const val = parseFloat(document.getElementById(`data-input-${i}`).value);

                const correctVal = currentProblem.inputs[i].correct;

                if (isNaN(val) || Math.abs(val - correctVal) > 0.01) {

                    feedback.className = 'feedback error';

                    feedback.innerText = `❌ Hiba az adatok kigyűjtésénél! Ellenőrizd a(z) ${currentProblem.inputs[i].name} értékét és a mértékegységet!`;

                    saveAttemptStats(currentStudent, diff, false);

                    return;

                }

            }

        }

 

        // 2. Végeredmény ellenőrzése

        const answer = parseFloat(document.getElementById('student-answer').value);

        if (isNaN(answer) || Math.abs(answer - currentProblem.target) > 0.05) {

            feedback.className = 'feedback error';

            feedback.innerText = '❌ A kiszámolt végeredmény nem pontos. Próbáld újra a képlet alapján!';

            saveAttemptStats(currentStudent, diff, false);

        } else {

            feedback.className = 'feedback success';

            feedback.innerText = '🎉 Helyes megoldás! Szép munka!';

            saveAttemptStats(currentStudent, diff, true);

        }

    }

 

    function saveAttemptStats(student, diff, isSuccess) {

        let stats = JSON.parse(localStorage.getItem('physics_stats') || '{}');

        if (!stats[student]) {

            stats[student] = { easy: 0, medium: 0, hard: 0, retries: 0 };

        }

 

        if (!isSuccess) {

            stats[student].retries += 1;

        } else {

            if (attempts === 1 || isSuccess) {

                stats[student][diff] += 1;

            }

        }

        localStorage.setItem('physics_stats', JSON.stringify(stats));

    }

 

    function verifyTeacher() {

        const pin = document.getElementById('teacher-pin').value;

        if (pin === '1234') {

            document.getElementById('teacher-login').style.display = 'none';

            document.getElementById('teacher-dashboard').style.display = 'block';

            loadTeacherStats();

        } else {

            alert('Hibás PIN kód! (Alapértelmezett: 1234)');

        }

    }

 

    function loadTeacherStats() {

        let stats = JSON.parse(localStorage.getItem('physics_stats') || '{}');

        const tbody = document.getElementById('stats-table-body');

        tbody.innerHTML = '';

 

        for (let student in stats) {

            let s = stats[student];

            let total = s.easy + s.medium + s.hard;

            let row = `<tr>

                <td><strong>${student}</strong></td>

                <td>${s.easy}</td>

                <td>${s.medium}</td>

                <td>${s.hard}</td>

                <td><strong style="color:var(--primary);">${total}</strong></td>

                <td><span style="color:var(--error);">${s.retries}</span></td>

            </tr>`;

            tbody.innerHTML += row;

        }

    }

 

    function exportCSV() {

        let stats = JSON.parse(localStorage.getItem('physics_stats') || '{}');

        let csv = 'Diak Neve,Konnyu,Kozepes,Nehez,Oszesal,Ujraprobalkozasok\n';

        for (let student in stats) {

            let s = stats[student];

            csv += `"${student}",${s.easy},${s.medium},${s.hard},${s.easy+s.medium+s.hard},${s.retries}\n`;

        }

        let blob = new Blob([csv], { type: 'text/csv' });

        let a = document.createElement('a');

        a.href = URL.createObjectURL(blob);

        a.download = 'fizika_haladas_eredmenyek.csv';

        a.click();

    }

 

    function clearData() {

        if (confirm('Biztosan törölni szeretnéd az összes diák eredményét?')) {

            localStorage.removeItem('physics_stats');

            loadTeacherStats();

        }

    }
